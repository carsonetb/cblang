expression: equality

equality: comparison ( ( "==" | "!=" ) comparison )*
comparison: term ( ( ">" | ">=" | "<" | "<=" ) tern )* ;
term: factor ( ( "+" | "-" ) factor )* ;
factor: unary ( ( "/" | "*" ) unary )* ;

unary:
    | ( "!" | "-" ) unary ;
    | primary ;

primary: 
    | INT
    | FLOAT
    | STRING
    | CHARACTER
    | TRUE_KW
    | FALSE_KW 
    | "(" expression ")" ;
