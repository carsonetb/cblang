program: 
    "class" "Main" ( "(" parameters? ")" )? "=" "(" members? ")"

parameters: IDENTIFIER template_definition? IDENTIFIER ( "," IDENTIFIER templates? IDENTIFIER )*
pass_params: expression ( "," expression )*
members: declaration ( "," declaration )*
templates: "<" ( IDENTIFIER templates? ( "," IDENTIFIER templates? ) )? ">"
template_definition: "<" IDENTIFIER ( "," IDENTIFIER )* ">"

declaration: 
    | varDecl
    | functionDecl
    | classDecl

varDecl: IDENTIFIER templates? IDENTIFIER ( "=" expression )?

functionDecl: "scope" IDENTIFIER template_definition? ( "(" parameters? ")" )? ( "->" IDENTIFIER )? "=" "{" scope? "}"

classDecl: "class" IDENTIFIER ( ":" IDENTIFIER ( "," IDENTIFIERS )* ) ( "(" parameters? ")" )? "=" "(" members? ")"

scope: statement ( statement )*

statement:
    ( expression 
    | IDENTIFIER = logic_or
    | IDENTIFIER templates? IDENTIFIER = expression
    | functionDecl )? ";"

logic_or: logic_and ( "||" logic_and )*
logic_and: euqality ( "&&" equality )*

expression: equality

equality: comparison ( ( "==" | "!=" ) comparison )*
comparison: term ( ( ">" | ">=" | "<" | "<=" ) tern )*
term: factor ( ( "+" | "-" ) factor )*
factor: unary ( ( "/" | "*" ) unary )*

unary:
    | ( "!" | "-" ) unary
    | primary

primary: 
    | INT
    | FLOAT
    | STRING
    | CHARACTER
    | TRUE_KW
    | FALSE_KW 
    | function_or_variable
    | "{" scope? "}" 
    | "(" logic_or ")"

function_or_variable: 
    | IDENTIFIER ( "(" pass_params? ")" )? ( "." function_or_variable )?
